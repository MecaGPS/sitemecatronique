Code Fonctionnement Capteur IR avec 2 moteurs

#include <#AccelStepper.h> #include <#Servo.h> // --- CONFIGURATION PINS --- const int pinCapteur = 3; const int SERVO_PIN = 11; // Moteur 1 (Tapis d'entrée) const int PUL1 = 9; const int DIR1 = 8; const int ENA1 = 10; // Moteur 2 (Tapis de sortie) const int PUL2 = 5; const int DIR2 = 4; const int ENA2 = 6; // --- INSTANCES --- AccelStepper tapis1(1, PUL1, DIR1); AccelStepper tapis2(1, PUL2, DIR2); Servo brasEjecteur; void setup() { pinMode(pinCapteur, INPUT); pinMode(ENA1, OUTPUT); pinMode(ENA2, OUTPUT); // Activation des drivers (ENA sur HIGH pour TB6600) digitalWrite(ENA1, HIGH); digitalWrite(ENA2, HIGH); // Configuration des moteurs tapis1.setMaxSpeed(2000); tapis2.setMaxSpeed(2000); // Configuration du Servo MG996R brasEjecteur.attach(SERVO_PIN); brasEjecteur.write(10); // Position repos sécurisée [cite: 47] Serial.begin(9600); Serial.println("Systeme de convoyage pret..."); } void loop() { // Lecture du capteur FC-51 : 0V (LOW) si obstacle détecté bool pieceDetectee = (digitalRead(pinCapteur) == LOW); if (!pieceDetectee) { // --- ÉTAPE 1 : RECHERCHE --- // Le tapis 1 avance, le tapis 2 attend tapis1.setSpeed(1000); tapis1.runSpeed(); tapis2.setSpeed(0); tapis2.runSpeed(); } else { // --- ÉTAPE 2 : DETECTION ET EJECTION --- Serial.println("Piece detectee !"); // 1. Arret immédiat du Moteur 1 tapis1.stop(); // 2. Action du bras ejecteur (MG996R) delay(500); brasEjecteur.write(70); // Angle d'éjection [cite: 47] delay(1000); // Temps de mouvement [cite: 14] brasEjecteur.write(10); // Retour au repos [cite: 47] delay(500); // --- ÉTAPE 3 : EVACUATION --- Serial.println("Lancement du tapis 2 (evacuation)..."); // On fait tourner le tapis 2 pendant 3 secondes unsigned long debutEvacuation = millis(); while(millis() - debutEvacuation < 3000) { tapis2.setSpeed(1000); tapis2.runSpeed(); } Serial.println("Cycle termine. Reprise du tapis 1."); delay(500); // Petite pause avant de relancer le cycle } }
