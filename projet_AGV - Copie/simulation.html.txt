      <h2>AGV 2 - Assistance opérateur</h2>
      <button onclick="envoyer('fournitures')">Aller au poste des fournitures</button>
      <button onclick="envoyer('outil')">Ramener un outil</button>
      <button onclick="envoyer('operateur')">Retour à l’opérateur</button>

  <script>
    let chaineActive = false;
    let intervalAGV1;

    function demarrerChaine() {
      document.getElementById("etatChaine").textContent = "État : en fonctionnement";
      chaineActive = true;
      deplacerAGV1();
    }

    function arreterChaine() {
      document.getElementById("etatChaine").textContent = "État : arrêté";
      chaineActive = false;
      clearInterval(intervalAGV1);
    }

    function deplacerAGV1() {
      const agv1 = document.getElementById("agv1");
      let position = 20;
      let direction = 1;
      intervalAGV1 = setInterval(() => {
        if (!chaineActive) return;
        position += direction * 2;
        agv1.style.left = position + "px";
        if (position > 420 || position < 20) direction *= -1;
      }, 30);
    }

    function envoyer(destination) {
      const etat = document.getElementById("etatAgv2");
      const agv = document.getElementById("agv2");

      etat.textContent = "AGV 2 se dirige vers " + destination + "...";

      let x, y;
      if (destination === "fournitures") { x = 50; y = 50; }
      else if (destination === "outil") { x = 400; y = 100; }
      else { x = 200; y = 300; }

      agv.style.left = x + "px";
      agv.style.top = y + "px";

      setTimeout(() => {
        etat.textContent = "AGV 2 est arrivé au poste : " + destination + ".";
      }, 2000);
    }
  </script>
